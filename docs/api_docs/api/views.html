 32@api_view(['POST'])
 33@permission_classes([permissions.AllowAny])
 34def auth_register(request):
 35    # Si viene con razon_social o acepto_terminos_y_privacidad, usar el serializer avanzado Ley 19.628
 36    if 'razon_social' in request.data or 'acepto_terminos_y_privacidad' in request.data or 'rut_empresa' in request.data:
 37        serializer = RegistroUsuarioEmpresaSerializer(data=request.data, context={'request': request})
 38        if serializer.is_valid():
 39            result = serializer.save()
 40            return Response({
 41                "mensaje": "Usuario y Empresa registrados exitosamente conforme a la Ley N° 19.628.",
 42                "usuario": result['user'].username,
 43                "empresa": result['empresa'].nombre
 44            }, status=status.HTTP_201_CREATED)
 45        return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)
 46    
 47    # Fallback para registro básico
 48    serializer = RegistroUsuarioSerializer(data=request.data)
 49    if serializer.is_valid():
 50        serializer.save()
 51        return Response({"mensaje": "Usuario creado exitosamente."}, status=status.HTTP_201_CREATED)
 52    return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)

 54@api_view(['POST'])

 56def solicitud_arco_view(request):
 57    """
 58    Permite a los usuarios autenticados solicitar la supresión, rectificación, acceso u oposición de sus datos (Ley 19.628).
 59    """
 60    tipo_derecho = request.data.get('tipo_derecho')
 61    detalles = request.data.get('detalles')
 62
 63    if not tipo_derecho or not detalles:
 64        return Response({'error': 'Debe especificar el tipo_derecho y los detalles de la solicitud.'}, status=status.HTTP_400_BAD_REQUEST)
 65
 66    solicitud = RegistroAuditoriaARCO.objects.create(
 67        usuario=request.user,
 68        tipo_derecho=tipo_derecho,
 69        detalles=detalles,
 70        estado='PENDIENTE'
 71    )
 72
 73    # Marcar flag en la empresa si aplica
 74    if hasattr(request.user, 'perfilusuario') and request.user.perfilusuario.empresa:
 75        empresa = request.user.perfilusuario.empresa
 76        empresa.solicitud_arco_activa = True
 77        empresa.save()
 78
 79    return Response({
 80        'mensaje': f'Solicitud de Derecho ARCO ({tipo_derecho}) registrada exitosamente.',
 81        'solicitud': RegistroAuditoriaARCOSerializer(solicitud).data
 82    }, status=status.HTTP_201_CREATED)

152    Endpoint POST /api/empresas/onboarding/
153    Recibe los datos del formulario multi-paso, guarda la empresa, ejecuta

178@api_view(['GET', 'PATCH'])
179@permission_classes([permissions.IsAuthenticated])
180def empresas_compliance(request, compliance_id=None):
181    """
182    Endpoint GET /api/empresas/compliance/ (lista todos)
183    Endpoint PATCH /api/empresas/compliance/<id>/ (actualiza responsable de una ley y tareas)
184    """
185    user = request.user
186    if not hasattr(user, 'perfilusuario') or not user.perfilusuario.empresa:
187        return Response([])
188    
189    empresa = user.perfilusuario.empresa
190
191    if request.method == 'PATCH' and compliance_id:
192        try:
193            compliance = ComplianceEmpresa.objects.get(id=compliance_id, empresa=empresa)
194            responsable = request.data.get('responsable')
195            
196            if responsable is not None:
197                compliance.responsable = responsable
198                compliance.save()
199                
200                # Cascade update to tasks
201                tareas = TareaPendiente.objects.filter(compliance_empresa=compliance)
202                tareas.update(responsable=responsable, responsable_asignado=responsable)
203                
204            return Response(ComplianceEmpresaSerializer(compliance).data)
205        except ComplianceEmpresa.DoesNotExist:
206            return Response({"error": "No encontrado"}, status=404)
207            
208    compliances = ComplianceEmpresa.objects.filter(empresa=empresa).select_related('normativa')
209    for comp in compliances:
210        comp.recalcular_progreso()
211        comp.save(update_fields=['porcentaje_progreso', 'estado'])
212    return Response(ComplianceEmpresaSerializer(compliances, many=True).data)

235class BaseEmpresaViewSet(viewsets.ModelViewSet):
236    permission_classes = [permissions.AllowAny]
237
238    def get_queryset(self):
239        user = self.request.user
240        if user.is_authenticated:
241            if user.is_superuser:
242                return self.queryset
243            if hasattr(user, 'perfilusuario'):
244                return self.queryset.filter(empresa=user.perfilusuario.empresa)
245        # Para el prototipo, si no está autenticado devolvemos todo
246        return self.queryset
247
248    def perform_create(self, serializer):
249        user = self.request.user
250        if user.is_authenticated and not user.is_superuser and hasattr(user, 'perfilusuario'):
251            serializer.save(empresa=user.perfilusuario.empresa)
252        else:
253            serializer.save()

362@api_view(['GET'])
363def recomendaciones_legales(request):
364    rubro = request.query_params.get('rubro')
365    tiene_datos = request.query_params.get('tiene_datos', 'false').lower() == 'true'
366    es_ecommerce = request.query_params.get('es_ecommerce', 'false').lower() == 'true'
367    tiene_residuos = request.query_params.get('tiene_residuos', 'false').lower() == 'true'
368
369    # Si el usuario estÃ¡ autenticado, inferir datos de su empresa
370    if request.user.is_authenticated and hasattr(request.user, 'perfilusuario') and request.user.perfilusuario.empresa:
371        empresa = request.user.perfilusuario.empresa
372        if not rubro:
373            rubro = empresa.rubro
374        tiene_datos = tiene_datos or empresa.maneja_datos_personales
375        es_ecommerce = es_ecommerce or empresa.es_b2c_ecommerce
376        tiene_residuos = tiene_residuos or empresa.genera_residuos_rep
377
378    leyes = ley_chile_client.buscar_normas_por_empresa(
379        rubro=rubro or 'TODOS',
380        tiene_datos=tiene_datos,
381        es_ecommerce=es_ecommerce,
382        tiene_residuos=tiene_residuos
383    )
384    return Response(leyes)
