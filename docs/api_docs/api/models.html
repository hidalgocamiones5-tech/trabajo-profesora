166    def recalcular_progreso(self):
167        """
168        Recalcula el porcentaje de progreso en cascada basado en las tareas y obligaciones.
169        """
170        # Buscar tareas vinculadas
171        tareas = self.tareas.all()
172        if not tareas.exists() and self.empresa and self.normativa:
173            from django.db.models import Q
174            cod = self.normativa.codigo_bcn or ''
175            alias = self.normativa.nombre or ''
176            tareas = TareaPendiente.objects.filter(
177                Q(compliance_empresa=self) |
178                Q(empresa=self.empresa, normativa=self.normativa) |
179                (Q(empresa=self.empresa) & (Q(asociada_a__icontains=cod) | Q(asociada_a__icontains=alias[:15])))
180            ).distinct()
181
182        if tareas.exists():
183            total = tareas.count()
184            completadas = tareas.filter(estado='completada').count()
185            self.porcentaje_progreso = round((completadas / total) * 100.0, 1)
186        else:
187            # Fallback a obligaciones
188            obligaciones = self.normativa.obligaciones.all()
189            if obligaciones.exists():
190                total = obligaciones.count()
191                cumplidas = obligaciones.filter(estado='cumplido').count()
192                self.porcentaje_progreso = round((cumplidas / total) * 100.0, 1)
193
194        if self.porcentaje_progreso >= 100.0 and self.estado in ['VERIFICADA', 'EN_PROCESO', 'ASIGNADA', 'PRELIMINAR']:
195            self.estado = 'CUMPLIDA'
196        elif self.porcentaje_progreso > 0 and self.estado in ['VERIFICADA', 'ASIGNADA', 'PRELIMINAR']:
197            self.estado = 'EN_PROCESO'
198
199        self.save(update_fields=['porcentaje_progreso', 'estado', 'updated_at'])
200
201        if self.normativa:
202            self.normativa.progreso = int(self.porcentaje_progreso)
203            if self.porcentaje_progreso >= 100:
204                self.normativa.estado = 'completada'
205            elif self.porcentaje_progreso > 0:
206                self.normativa.estado = 'en_tiempo'
207            self.normativa.save(update_fields=['progreso', 'estado'])
208
209        return self.porcentaje_progreso
