 31    def auditar_rag_view(self, request, object_id):
 32        empresa = self.get_object(request, object_id)
 33        if empresa:
 34            servicio = GrcAuditService()
 35            res = servicio.auditar_y_generar_borrador(empresa)
 36            if res.get("success"):
 37                registro_id = res.get("registro_id")
 38                messages.info(
 39                    request,
 40                    f"🤖 Diagnóstico IA generado para {empresa.nombre}. Por favor revisa y aprueba las normativas y tareas asignadas."
 41                )
 42                return redirect('admin:rag_admin_registroauditoriarag_revisar', object_id=registro_id)
 43            else:
 44                self.message_user(request, f"❌ Error ejecutando RAG: {res.get('error')}", level="ERROR")
 45        return redirect('admin:rag_admin_auditoriaiaempresa_changelist')

 47    def ejecutar_rag_masivo(self, request, queryset):
 48        servicio = GrcAuditService()
 49        auditadas = 0
 50        ultimo_registro_id = None

 61        self.message_user(request, f"🤖 Auditoría RAG masiva generada: {auditadas} borradores creados para revisión.", level="SUCCESS")
